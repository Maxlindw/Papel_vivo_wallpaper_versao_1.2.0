cmake_minimum_required(VERSION 3.20)
project(PapelVivo VERSION 2.0.1 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PAPEL_BUILD_TESTS "Compilar testes puros de política" ON)
if(PAPEL_BUILD_TESTS)
 enable_testing()
 add_executable(papel_policy_tests tests/policy_test.cpp)
 target_include_directories(papel_policy_tests PRIVATE include)
 add_test(NAME politica COMMAND papel_policy_tests)
 add_executable(papel_workflow_tests tests/workflow_test.cpp)
 target_include_directories(papel_workflow_tests PRIVATE include)
 add_test(NAME correcao COMMAND papel_workflow_tests)
endif()
if(WIN32)
 enable_language(RC)
 add_executable(PapelVivo WIN32 src/supervisor.cpp assets/app.rc)
 target_include_directories(PapelVivo PRIVATE include)
 target_compile_definitions(PapelVivo PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 target_link_libraries(PapelVivo PRIVATE user32 shell32)
 if(MINGW)
 target_link_options(PapelVivo PRIVATE -municode -static)
 endif()
 install(TARGETS PapelVivo RUNTIME DESTINATION .)
 add_executable(PapelVivoCore WIN32 src/main.cpp src/player.cpp src/media_job.cpp assets/app.rc)
 set_target_properties(PapelVivoCore PROPERTIES OUTPUT_NAME "PapelVivo.Core")
 target_include_directories(PapelVivoCore PRIVATE include vendor/mpv vendor/json)
 target_compile_definitions(PapelVivoCore PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
 target_link_libraries(PapelVivoCore PRIVATE user32 gdi32 comctl32 comdlg32 shell32 ole32 opengl32 dwmapi wtsapi32 psapi uuid gdiplus uxtheme)
 if(MSVC)
  target_compile_options(PapelVivoCore PRIVATE /W4 /utf-8 /permissive-)
  set_property(TARGET PapelVivoCore PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 else()
  target_compile_options(PapelVivoCore PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers -finput-charset=UTF-8)
  target_link_options(PapelVivoCore PRIVATE -municode -static -static-libgcc -static-libstdc++)
 endif()
 install(TARGETS PapelVivoCore RUNTIME DESTINATION .)
endif()


if(PAPEL_BUILD_TESTS)
 add_executable(player_catalog_tests tests/catalog_test.cpp)
 target_include_directories(player_catalog_tests PRIVATE include vendor/json)
 if(MINGW)
 target_link_options(player_catalog_tests PRIVATE -static)
 endif()
 add_test(NAME gallery COMMAND player_catalog_tests)
endif()
